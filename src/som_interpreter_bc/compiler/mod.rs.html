20#[derive(Debug, Clone)]
21pub enum Literal {
22    Symbol(Interned),
23    String(Gc<String>),
24    Double(f64),
25    Integer(i32),
26    BigInteger(Gc<BigInt>),
27    Array(VecLiteral),
28    Block(Gc<MethodInfo>),
29}
30
31impl PartialEq for Literal {
32    fn eq(&self, other: &Self) -> bool {
33        match (self, other) {
34            (Literal::Symbol(val1), Literal::Symbol(val2)) => val1.eq(val2),
35            (Literal::String(val1), Literal::String(val2)) => val1.eq(val2),
36            (Literal::Double(val1), Literal::Double(val2)) => val1.eq(val2),
37            (Literal::Integer(val1), Literal::Integer(val2)) => val1.eq(val2),
38            (Literal::BigInteger(val1), Literal::BigInteger(val2)) => val1.eq(val2),
39            (Literal::Array(val1), Literal::Array(val2)) => val1.eq(val2),
40            (Literal::Block(val1), Literal::Block(val2)) => val1 == val2,
41            _ => false,
42        }
43    }
44}
45
46impl Eq for Literal {}
47
48impl Hash for Literal {
49    fn hash<H: Hasher>(&self, state: &mut H) {
50        match self {
51            Literal::Symbol(val) => {
52                state.write(b"sym#");
53                val.hash(state);
54            }
55            Literal::String(val) => {
56                state.write(b"string#");
57                val.hash(state);
58            }
59            Literal::Double(val) => {
60                state.write(b"dbl#");
61                val.to_bits().hash(state);
62            }
63            Literal::Integer(val) => {
64                state.write(b"int#");
65                val.hash(state);
66            }
67            Literal::BigInteger(val) => {
68                state.write(b"bigint#");
69                val.hash(state);
70            }
71            Literal::Array(val) => {
72                state.write(b"array#");
73                for elem in val.iter() {
74                    elem.hash(state)
75                }
76            }
77            Literal::Block(val) => {
78                state.write(b"blk");
79                val.hash(state);
80            }
81        }
82    }
83}
84
85pub fn value_from_literal(literal: &Literal, gc_interface: &mut GCInterface) -> Value {
86    match literal {
87        Literal::Symbol(sym) => Value::Symbol(*sym),
88        Literal::String(val) => Value::String(val.clone()),
89        Literal::Double(val) => Value::Double(*val),
90        Literal::Integer(val) => Value::Integer(*val),
91        Literal::BigInteger(val) => Value::BigInteger(val.clone()),
92        Literal::Array(val) => {
93            let arr = &val.iter().map(|lit| value_from_literal(lit, gc_interface)).collect::<Vec<_>>();
94            Value::Array(VecValue(gc_interface.alloc_slice(arr, AllocSiteMarker::VecValue)))
95        }
96        Literal::Block(_) => {
97            panic!("We should never request a value from a block literal, which are always only associated with a PushBlock bytecode")
98        }
99    }
100}
