25#[derive(Debug, Clone, PartialEq)]
26pub struct TrivialGetterMethod {
27    pub(crate) field_idx: u8,
28}
29
30impl Invoke for TrivialGetterMethod {
31    fn invoke(&mut self, _universe: &mut Universe, value_stack: &mut GlobalValueStack, nbr_args: usize) -> Return {
32        debug_assert_eq!(nbr_args, 1);
33        let arg = value_stack.pop();
34
35        if let Some(cls) = arg.as_class() {
36            Return::Local(cls.class().lookup_field(self.field_idx))
37        } else if let Some(instance) = arg.as_instance() {
38            Return::Local(*Instance::lookup_field(&instance, self.field_idx))
39        } else {
40            panic!("trivial getter not called on a class/instance?")
41        }
42    }
43}

45#[derive(Debug, Clone, PartialEq)]
46pub struct TrivialSetterMethod {
47    pub(crate) field_idx: u8,
48}
49
50impl Invoke for TrivialSetterMethod {
51    fn invoke(&mut self, _universe: &mut Universe, value_stack: &mut GlobalValueStack, nbr_args: usize) -> Return {
52        debug_assert_eq!(nbr_args, 2);
53        let val = value_stack.pop();
54        let rcvr = value_stack.pop();
55
56        if let Some(cls) = rcvr.as_class() {
57            cls.class().assign_field(self.field_idx, val);
58            Return::Local(Value::Class(cls))
59        } else if let Some(instance) = rcvr.as_instance() {
60            Instance::assign_field(&instance, self.field_idx, val);
61            Return::Local(Value::Instance(instance))
62        } else {
63            panic!("trivial getter not called on a class/instance?")
64        }
65    }
66}
