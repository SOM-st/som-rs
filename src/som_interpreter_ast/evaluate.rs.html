16/// The trait for evaluating AST nodes.
17pub trait Evaluate {
18    /// Evaluate the node within a given universe.
19    fn evaluate(&mut self, universe: &mut Universe, value_stack: &mut GlobalValueStack) -> Return;
20}

22impl Evaluate for AstExpression {
23    fn evaluate(&mut self, universe: &mut Universe, value_stack: &mut GlobalValueStack) -> Return {
24        match self {
25            Self::LocalVarWrite(idx, expr) => {
26                let value = propagate!(expr.evaluate(universe, value_stack));
27                universe.current_frame.assign_local(*idx, value);
28                Return::Local(value)
29            }
30            Self::NonLocalVarWrite(scope, idx, expr) => {
31                let value = propagate!(expr.evaluate(universe, value_stack));
32                Frame::nth_frame_back(&universe.current_frame, *scope).assign_local(*idx, value);
33                Return::Local(value)
34            }
35            Self::FieldWrite(idx, expr) => {
36                let value = propagate!(expr.evaluate(universe, value_stack));
37                universe.current_frame.assign_field(*idx, &value);
38                Return::Local(value)
39            }
40            Self::ArgWrite(scope, idx, expr) => {
41                let value = propagate!(expr.evaluate(universe, value_stack));
42                Frame::nth_frame_back(&universe.current_frame, *scope).assign_arg(*idx, value);
43                Return::Local(value)
44            }
45            Self::IncLocal(idx) => {
46                let local_val = universe.current_frame.lookup_local_mut(*idx);
47                if let Some(int) = local_val.as_integer() {
48                    *local_val = Value::new_integer(int + 1);
49                } else if let Some(double) = local_val.as_double() {
50                    *local_val = Value::new_double(double + 1.0);
51                } else if let Some(mut big_int) = local_val.as_big_integer::<Gc<BigInt>>() {
52                    *big_int += 1;
53                } else {
54                    panic!("Invalid type in Inc: {:?}", local_val)
55                }
56                Return::Local(*local_val)
57            }
58            Self::DecLocal(idx) => {
59                let local_val = universe.current_frame.lookup_local_mut(*idx);
60                if let Some(int) = local_val.as_integer() {
61                    *local_val = Value::new_integer(int - 1);
62                } else if let Some(double) = local_val.as_double() {
63                    *local_val = Value::new_double(double - 1.0);
64                } else if let Some(mut big_int) = local_val.as_big_integer::<Gc<BigInt>>() {
65                    *big_int -= 1;
66                } else {
67                    panic!("Invalid type in Dec: {:?}", local_val)
68                }
69                Return::Local(*local_val)
70            }
71            Self::Block(blk) => blk.evaluate(universe, value_stack),
72            Self::LocalExit(expr) => {
73                let value = propagate!(expr.evaluate(universe, value_stack));
74                Return::NonLocal(value, universe.current_frame.clone())
75            }
76            Self::NonLocalExit(expr, scope) => {
77                debug_assert_ne!(*scope, 0);
78
79                let value = propagate!(expr.evaluate(universe, value_stack));
80                let method_frame = Frame::nth_frame_back(&universe.current_frame, *scope);
81
82                let has_not_escaped = {
83                    let mut current_frame = universe.current_frame.clone();
84
85                    loop {
86                        if current_frame == method_frame {
87                            break true;
88                        } else if current_frame.is_empty() {
89                            break false;
90                        } else {
91                            current_frame = current_frame.prev_frame.clone();
92                        }
93                    }
94                };
95
96                if has_not_escaped {
97                    // the BC interp has to pop all the escaped frames here, we don't (because we chain return nonlocals, exception-style?).
98                    Return::NonLocal(value, method_frame)
99                } else {
100                    // Block has escaped its method frame.
101                    let instance = method_frame.get_self();
102                    let frame = &universe.current_frame;
103                    let block = match frame.lookup_argument(0).as_block() {
104                        Some(blk) => blk,
105                        _ => {
106                            // Should never happen, because `universe.current_frame()` would
107                            // have been equal to `universe.current_method_frame()`.
108                            panic!("A method frame has escaped itself ??");
109                        }
110                    };
111                    universe
112                        .escaped_block(value_stack, instance, block)
113                        .unwrap_or_else(|| panic!("A block has escaped and `escapedBlock:` is not defined on receiver"))
114                }
115            }
116            Self::Literal(literal) => literal.evaluate(universe, value_stack),
117            Self::LocalVarRead(idx) => {
118                let local = universe.current_frame.lookup_local(*idx);
119                Return::Local(*local)
120            }
121            Self::NonLocalVarRead(scope, idx) => {
122                let non_local = *Frame::nth_frame_back(&universe.current_frame, *scope).lookup_local(*idx);
123                Return::Local(non_local)
124            }
125            Self::FieldRead(idx) => {
126                let field = universe.current_frame.lookup_field(*idx);
127                Return::Local(field)
128            }
129            Self::ArgRead(scope, idx) => {
130                let arg = *Frame::nth_frame_back(&universe.current_frame, *scope).lookup_argument(*idx);
131                Return::Local(arg)
132            }
133            Self::GlobalRead(global_node) => global_node.evaluate(universe, value_stack),
134            Self::UnaryDispatch(un_op) => un_op.evaluate(universe, value_stack),
135            Self::BinaryDispatch(bin_op) => bin_op.evaluate(universe, value_stack),
136            Self::TernaryDispatch(ter_op) => ter_op.evaluate(universe, value_stack),
137            Self::NAryDispatch(msg) => msg.evaluate(universe, value_stack),
138            Self::SuperMessage(msg) => msg.evaluate(universe, value_stack),
139            Self::InlinedCall(inlined_node) => match inlined_node.as_mut() {
140                InlinedNode::IfInlined(if_inlined) => if_inlined.evaluate(universe, value_stack),
141                InlinedNode::IfTrueIfFalseInlined(if_true_if_false_inlined) => if_true_if_false_inlined.evaluate(universe, value_stack),
142                InlinedNode::WhileInlined(while_inlined) => while_inlined.evaluate(universe, value_stack),
143                InlinedNode::OrInlined(or_inlined) => or_inlined.evaluate(universe, value_stack),
144                InlinedNode::AndInlined(and_inlined) => and_inlined.evaluate(universe, value_stack),
145                InlinedNode::ToDoInlined(to_do_inlined) => to_do_inlined.evaluate(universe, value_stack),
146                InlinedNode::IfNilInlined(if_nil_inlined_node) => if_nil_inlined_node.evaluate(universe, value_stack),
147                InlinedNode::IfNilIfNotNilInlined(if_nil_if_not_nil_inlined_node) => if_nil_if_not_nil_inlined_node.evaluate(universe, value_stack),
148            },
149        }
150    }
151}

153impl Evaluate for AstLiteral {
154    fn evaluate(&mut self, universe: &mut Universe, _value_stack: &mut GlobalValueStack) -> Return {
155        match self {
156            Self::Array(array) => {
157                // todo: couldn't we precompute those astliterals, really?
158                let mut output = Vec::with_capacity(array.len());
159                for literal in array.iter() {
160                    let value = propagate!(literal.clone().evaluate(universe, _value_stack));
161                    output.push(value);
162                }
163                Return::Local(Value::Array(VecValue(
164                    universe.gc_interface.alloc_slice(&output, AllocSiteMarker::VecValue),
165                )))
166            }
167            Self::Integer(int) => Return::Local(Value::Integer(*int)),
168            Self::BigInteger(bigint) => Return::Local(Value::BigInteger(bigint.clone())),
169            Self::Double(double) => Return::Local(Value::Double(*double)),
170            Self::Symbol(sym) => Return::Local(Value::Symbol(*sym)),
171            Self::String(string) => Return::Local(Value::String(string.clone())),
172        }
173    }
174}

182impl Evaluate for Gc<AstBlock> {
183    fn evaluate(&mut self, universe: &mut Universe, _value_stack: &mut GlobalValueStack) -> Return {
184        debug_assert_valid_semispace_ptr!(self);
185        let mut block_ptr = universe.gc_interface.request_memory_for_type(size_of::<Block>(), som_gc::gc_interface::AllocSiteMarker::RuntimeBlock);
186        *block_ptr = Block {
187            block: self.clone(),
188            frame: universe.current_frame.clone(),
189        };
190        Return::Local(Value::Block(block_ptr))
191    }
192}

194impl AstDispatchNode {
195    #[inline(always)]
196    fn lookup_and_dispatch(&mut self, nbr_args: usize, universe: &mut Universe, value_stack: &mut GlobalValueStack) -> Return {
197        let receiver = *value_stack.iter().nth_back(nbr_args - 1).unwrap();
198
199        let invokable = match &mut self.inline_cache {
200            Some((cached_rcvr_ptr, ref mut method)) => {
201                debug_assert_valid_semispace_ptr!(method);
202
203                if *cached_rcvr_ptr == receiver.class(universe) {
204                    return method.invoke(universe, value_stack, nbr_args);
205                } else {
206                    receiver.lookup_method(universe, self.signature)
207                }
208            }
209            None => receiver.lookup_method(universe, self.signature),
210        };
211        // let invokable = receiver.lookup_method(universe, &self.signature);
212
213        match invokable {
214            Some(mut invokable) => {
215                debug_assert_valid_semispace_ptr_value!(receiver);
216                let receiver_class_ref = receiver.class(universe);
217                debug_assert_valid_semispace_ptr!(receiver_class_ref);
218                self.inline_cache = Some((receiver_class_ref, invokable.clone()));
219
220                invokable.invoke(universe, value_stack, nbr_args)
221            }
222            None => {
223                let mut args = value_stack.split_off_n(nbr_args);
224                let receiver = args.remove(0);
225                universe.does_not_understand(value_stack, receiver, self.signature, args).unwrap_or_else(|| {
226                    panic!(
227                        "could not find method '{}>>#{}'",
228                        receiver.class(universe).name(),
229                        universe.lookup_symbol(self.signature)
230                    )
231                })
232            }
233        }
234    }
235}

273impl Evaluate for AstNAryDispatch {
274    fn evaluate(&mut self, universe: &mut Universe, value_stack: &mut GlobalValueStack) -> Return {
275        let receiver = propagate!(self.dispatch_node.receiver.evaluate(universe, value_stack));
276
277        value_stack.push(receiver);
278
279        for expr in &mut self.values {
280            let value = propagate!(expr.evaluate(universe, value_stack));
281            value_stack.push(value);
282        }
283
284        debug_assert!(
285            self.values.len() > 2,
286            "should be a specialized unary/binary/ternary node, not a generic N-ary node"
287        );
288
289        self.dispatch_node.lookup_and_dispatch(self.values.len() + 1, universe, value_stack)
290    }
291}

293impl Evaluate for AstSuperMessage {
294    fn evaluate(&mut self, universe: &mut Universe, value_stack: &mut GlobalValueStack) -> Return {
295        let invokable = self.super_class.lookup_method(self.signature);
296        let receiver = universe.current_frame.get_self();
297        value_stack.push(receiver);
298
299        for expr in &mut self.values {
300            let value = propagate!(expr.evaluate(universe, value_stack));
301            value_stack.push(value);
302        }
303
304        match invokable {
305            Some(mut invokable) => invokable.invoke(universe, value_stack, self.values.len() + 1),
306            None => {
307                let mut args = value_stack.split_off_n(self.values.len() + 1);
308                let receiver = args.remove(0);
309                universe.does_not_understand(value_stack, receiver, self.signature, args).unwrap_or_else(|| {
310                    panic!(
311                        "could not find method '{}>>#{}'",
312                        receiver.class(universe).name(),
313                        universe.lookup_symbol(self.signature)
314                    )
315                    // Return::Local(Value::Nil)
316                })
317            }
318        }
319    }
320}

322impl Evaluate for AstBody {
323    fn evaluate(&mut self, universe: &mut Universe, value_stack: &mut GlobalValueStack) -> Return {
324        let mut last_value = Value::NIL;
325        for expr in &mut self.exprs {
326            last_value = propagate!(expr.evaluate(universe, value_stack));
327        }
328        Return::Local(last_value)
329    }

332impl Evaluate for AstMethodDef {
333    fn evaluate(&mut self, universe: &mut Universe, value_stack: &mut GlobalValueStack) -> Return {
334        // HACK: we want to hold on to a reference to the current frame at that point in time.
335        // We can't copy/clone the pointer, because we want to have a reference to that pointer in case moving GC moves it.
336        // And we can't hold onto an immutable ref to universe while passing a mutable ref universe to `self.body.evaluate` lower down. Hence this hack.
337        // Not sure how to better solve that one, to be honest.
338        let current_frame = unsafe { &*(&universe.current_frame as *const Gc<Frame>) };
339
340        loop {
341            match self.body.evaluate(universe, value_stack) {
342                Return::NonLocal(value, frame) => {
343                    if *current_frame == frame {
344                        break Return::Local(value);
345                    } else {
346                        break Return::NonLocal(value, frame);
347                    }
348                }
349                Return::Local(_) => break Return::Local(current_frame.get_self()),
350                Return::Restart => continue,
351            }
352        }
353    }
354}
