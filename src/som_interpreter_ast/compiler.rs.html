84    pub fn find_var(&self, name: &str, cur_scope: usize, scopes: &Vec<AstScopeCtxt>) -> Option<FoundVar> {
85        let name = match name {
86            "super" => "self",
87            name => name,
88        };
89        (self.locals.get_index_of(name))
90            .map(|idx| FoundVar::Local(0, idx as u8))
91            .or_else(|| (self.args.get_index_of(name)).map(|idx| FoundVar::Argument(0, idx as u8)))
92            .or_else(|| {
93                scopes.iter().nth_back(cur_scope + 1)?.find_var(name, cur_scope + 1, scopes).map(|found| match found {
94                    FoundVar::Local(up_idx, idx) => FoundVar::Local(up_idx + 1, idx),
95                    FoundVar::Argument(up_idx, idx) => FoundVar::Argument(up_idx + 1, idx),
96                    _ => unreachable!(),
97                })
98            })
99    }

103    fn find_var(&self, name: &str) -> Option<FoundVar> {
104        if let Some(found_var) = self.scopes.last()?.find_var(name, 0, &self.scopes) {
105            return Some(found_var);
106        }
107        if let Some(cls) = &self.class {
108            if let Some(found_field) = cls.field_names.iter().position(|n| n == name) {
109                return Some(FoundVar::Field(found_field as u8));
110            }
111        }
112        None
113    }

124    pub fn get_method_kind(method: &ast::MethodDef, class: Option<Gc<Class>>, gc_interface: &mut GCInterface, interner: &mut Interner) -> MethodKind {
125        match method.body {
126            MethodBody::Primitive => MethodKind::Primitive(&*UNIMPLEM_PRIMITIVE),
127            MethodBody::Body { .. } => {
128                let ast_method_def = AstMethodCompilerCtxt::parse_method_def(method, class, gc_interface, interner);
129
130                // since we allocate the method, we report the size of the method to GC, which is enough for trivial methods
131                if let Some(trivial_method_kind) = AstMethodCompilerCtxt::make_trivial_method_if_possible(&ast_method_def, interner) {
132                    trivial_method_kind
133                } else {
134                    // ...but for regular methods, we calculate the total size of the expressions also
135                    // we only add the size of the expressions and not the vector since the vector is already counted within the methodkind
136
137                    // NOT the best way to do this, but I thought it'd be nice to do it AFTER all the compilation and inlining is done, to make sure we are accurate
138                    #[cfg(feature = "track-allocations")]
139                    {
140                        fn get_tree_size(expr: &AstExpression) -> u128 {
141                            use AstExpression::*;
142                            let expr_size = size_of::<AstExpression>() as u128;
143                            match expr {
144                                LocalVarRead(_)
145                                | NonLocalVarRead(_, _)
146                                | ArgRead(_, _)
147                                | FieldRead(_)
148                                | IncLocal(_)
149                                | DecLocal(_)
150                                | GlobalRead(_)
151                                | Literal(_) => expr_size,
152                                LocalVarWrite(_, ast_expression)
153                                | NonLocalVarWrite(_, _, ast_expression)
154                                | ArgWrite(_, _, ast_expression)
155                                | LocalExit(ast_expression)
156                                | NonLocalExit(ast_expression, _)
157                                | FieldWrite(_, ast_expression) => expr_size + get_tree_size(ast_expression),
158                                UnaryDispatch(ast_unary_dispatch) => {
159                                    // the minus feels silly, but the logic is that since it's not a box in AstUnaryDispatch,
160                                    // if we count the size of AstUnaryDispatch but then also invoke get_tree_size on the receiver, the receiver would get counted twice
161                                    expr_size + size_of::<AstUnaryDispatch>() as u128 + get_tree_size(&ast_unary_dispatch.dispatch_node.receiver)
162                                        - expr_size
163                                }
164                                BinaryDispatch(ast_binary_dispatch) => {
165                                    // here count an AstBinaryDispatch but avoid double counting the expressions
166                                    expr_size
167                                        + size_of::<AstBinaryDispatch>() as u128
168                                        + get_tree_size(&ast_binary_dispatch.arg)
169                                        + get_tree_size(&ast_binary_dispatch.dispatch_node.receiver)
170                                        - expr_size * 2
171                                }
172                                TernaryDispatch(ast_ternary_dispatch) => {
173                                    expr_size
174                                        + size_of::<AstTernaryDispatch>() as u128
175                                        + get_tree_size(&ast_ternary_dispatch.dispatch_node.receiver)
176                                        + get_tree_size(&ast_ternary_dispatch.arg1)
177                                        + get_tree_size(&ast_ternary_dispatch.arg2)
178                                        - expr_size * 3
179                                }
180                                NAryDispatch(ast_nary_dispatch) => {
181                                    expr_size
182                                        + size_of::<AstNAryDispatch>() as u128
183                                        + get_tree_size(&ast_nary_dispatch.dispatch_node.receiver)
184                                        + ast_nary_dispatch.values.iter().map(get_tree_size).sum::<u128>()
185                                        - expr_size // only one here - it's a Vec (counted within ASTNaryDispatch), so we've not double counted elements
186                                }
187                                SuperMessage(ast_super_message) => {
188                                    expr_size
189                                        + size_of::<AstSuperMessage>() as u128
190                                        + ast_super_message.values.iter().map(get_tree_size).sum::<u128>()
191                                }
192                                Block(block) => {
193                                    // block is on the GC heap, we only need to consider its body expressions.
194                                    expr_size + block.body.exprs.iter().map(get_tree_size).sum::<u128>()
195                                }
196                                InlinedCall(inlined_node) => {
197                                    use crate::ast::InlinedNode;
198                                    use crate::ast::InlinedNode::*;
199                                    expr_size + size_of::<InlinedNode>() as u128 + {
200                                        match &**inlined_node {
201                                            IfInlined(if_inlined_node) => if_inlined_node.body_instrs.exprs.iter().map(get_tree_size).sum::<u128>(),
202                                            IfTrueIfFalseInlined(if_true_if_false_inlined_node) => {
203                                                get_tree_size(&if_true_if_false_inlined_node.cond_expr)
204                                                    + if_true_if_false_inlined_node.body_1_instrs.exprs.iter().map(get_tree_size).sum::<u128>()
205                                                    + if_true_if_false_inlined_node.body_2_instrs.exprs.iter().map(get_tree_size).sum::<u128>()
206                                            }
207                                            IfNilInlined(if_nil_inlined_node) => {
208                                                get_tree_size(&if_nil_inlined_node.cond_expr)
209                                                    + if_nil_inlined_node.body_instrs.exprs.iter().map(get_tree_size).sum::<u128>()
210                                            }
211                                            IfNilIfNotNilInlined(if_nil_if_not_nil_inlined_node) => {
212                                                get_tree_size(&if_nil_if_not_nil_inlined_node.cond_expr)
213                                                    + if_nil_if_not_nil_inlined_node.body_1_instrs.exprs.iter().map(get_tree_size).sum::<u128>()
214                                                    + if_nil_if_not_nil_inlined_node.body_2_instrs.exprs.iter().map(get_tree_size).sum::<u128>()
215                                            }
216                                            WhileInlined(while_inlined_node) => {
217                                                while_inlined_node.cond_instrs.exprs.iter().map(get_tree_size).sum::<u128>()
218                                                    + while_inlined_node.body_instrs.exprs.iter().map(get_tree_size).sum::<u128>()
219                                            }
220                                            OrInlined(or_inlined_node) => {
221                                                get_tree_size(&or_inlined_node.first)
222                                                    + or_inlined_node.second.exprs.iter().map(get_tree_size).sum::<u128>()
223                                            }
224                                            AndInlined(and_inlined_node) => {
225                                                get_tree_size(&and_inlined_node.first)
226                                                    + and_inlined_node.second.exprs.iter().map(get_tree_size).sum::<u128>()
227                                            }
228                                            ToDoInlined(to_do_inlined_node) => {
229                                                get_tree_size(&to_do_inlined_node.start)
230                                                    + get_tree_size(&to_do_inlined_node.end)
231                                                    + to_do_inlined_node.body.exprs.iter().map(get_tree_size).sum::<u128>()
232                                            }
233                                        }
234                                    }
235                                }
236                            }
237                        }
238                        let total_tree_size: u128 = ast_method_def.body.exprs.iter().map(get_tree_size).sum();
239                        gc_interface.total_program_repr_size += total_tree_size;
240                    }
241                    MethodKind::Defined(ast_method_def)
242                }
243            }
244        }
245        // match method.signature.as_str() {
246        //     // "to:by:do:" => MethodKind::Specialized(MethodKindSpecialized::ToByDo(ToByDoNode {})),
247        //     // "downTo:do:" => MethodKind::Specialized(MethodKindSpecialized::DownToDo(DownToDoNode {})),
248        //     _ => match method.body {

251    pub(crate) fn make_trivial_method_if_possible(method_def: &AstMethodDef, _interner: &mut Interner) -> Option<MethodKind> {
252        if method_def.locals_nbr != 0 || method_def.body.exprs.len() != 1 {
253            return None;
254        }
255
256        let args_nbr = method_def.signature.chars().filter(|e| *e == ':').count();
257
258        match method_def.body.exprs.first()? {
259            AstExpression::LocalExit(expr) => {
260                if args_nbr != 0 {
261                    return None;
262                }
263
264                match expr.as_ref() {
265                    AstExpression::Literal(lit) => {
266                        Some(MethodKind::TrivialLiteral(TrivialLiteralMethod { literal: lit.clone() }))
267                        // todo avoid clone by moving code to previous function tbh
268                    }
269                    AstExpression::GlobalRead(global) => Some(MethodKind::TrivialGlobal(TrivialGlobalMethod { global_name: global.clone() })),
270                    AstExpression::FieldRead(idx) => Some(MethodKind::TrivialGetter(TrivialGetterMethod { field_idx: *idx })),
271                    _ => None,
272                }
273            }
274            AstExpression::FieldWrite(idx, expr) => {
275                if args_nbr != 1 {
276                    return None;
277                }
278
279                match expr.as_ref() {
280                    AstExpression::ArgRead(0, 1) => Some(MethodKind::TrivialSetter(TrivialSetterMethod { field_idx: *idx })),
281                    _ => None,
282                }
283            }
284            _ => None,
285        }
286    }

288    /// Transforms a generic MethodDef into an AST-specific one.
289    /// Note: public since it's used in tests.
290    pub fn parse_method_def(
291        method_def: &ast::MethodDef,
292        class: Option<Gc<Class>>,
293        gc_interface: &mut GCInterface,
294        interner: &mut Interner,
295    ) -> AstMethodDef {
296        let (body, locals_nbr) = match &method_def.body {
297            MethodBody::Primitive => {
298                unreachable!("unimplemented primitive")
299            }
300            MethodBody::Body { locals_nbr, body, locals } => {
301                let args_nbr = method_def.signature.chars().filter(|e| *e == ':').count(); // not sure if needed
302
303                let mut locals_set: IndexSet<String> = IndexSet::new();
304                for local in locals {
305                    locals_set.insert(local.clone());
306                }
307
308                let mut args_set: IndexSet<String> = IndexSet::new();
309                args_set.insert("self".to_string());
310                for arg in &method_def.args {
311                    args_set.insert(arg.clone());
312                }
313                let mut ctxt = AstMethodCompilerCtxt {
314                    class,
315                    scopes: vec![AstScopeCtxt::init(args_nbr, *locals_nbr, locals_set, args_set)],
316                    gc_interface,
317                    interner,
318                };
319
320                (ctxt.parse_body(body), ctxt.scopes.last().unwrap().get_nbr_locals() as u8)
321            }
322        };
323
324        AstMethodDef {
325            signature: method_def.signature.clone(),
326            locals_nbr,
327            body,
328        }
329    }

331    pub fn parse_expression(&mut self, expr: &Expression) -> AstExpression {
332        match expr.clone() {
333            Expression::Read(global_name) => match self.find_var(&global_name) {
334                Some(FoundVar::Local(scope, idx)) => match scope {
335                    0 => AstExpression::LocalVarRead(idx),
336                    _ => AstExpression::NonLocalVarRead(scope, idx),
337                },
338                Some(FoundVar::Argument(scope, idx)) => AstExpression::ArgRead(scope, idx),
339                Some(FoundVar::Field(idx)) => AstExpression::FieldRead(idx),
340                None => self.global_read(global_name),
341            },
342            Expression::Write(global_name, expr) => match self.find_var(&global_name) {
343                Some(FoundVar::Local(scope, idx)) => match scope {
344                    0 => {
345                        let local_write_expr = AstExpression::LocalVarWrite(idx, Box::new(self.parse_expression(expr.as_ref())));
346                        self.maybe_make_inc_or_dec(&local_write_expr).unwrap_or(local_write_expr)
347                    }
348                    _ => AstExpression::NonLocalVarWrite(scope, idx, Box::new(self.parse_expression(expr.as_ref()))),
349                },
350                Some(FoundVar::Argument(scope, idx)) => AstExpression::ArgWrite(scope, idx, Box::new(self.parse_expression(expr.as_ref()))),
351                Some(FoundVar::Field(idx)) => AstExpression::FieldWrite(idx, Box::new(self.parse_expression(expr.as_ref()))),
352                _ => self.resolve_global_write(&global_name, &expr),
353            },
354            Expression::Message(msg) => self.parse_message(msg.as_ref()),
355            Expression::Exit(expr) => {
356                let scope = self.scopes.len() - 1;
357
358                match scope {
359                    0 => AstExpression::LocalExit(Box::new(self.parse_expression(expr.as_ref()))),
360                    _ => AstExpression::NonLocalExit(Box::new(self.parse_expression(expr.as_ref())), scope as u8),
361                }
362            }
363            Expression::Literal(a) => {
364                match &a {
365                    // this is to handle a weird corner case where "-2147483648" is considered to be a bigint by the lexer and then parser, when it's in fact just barely in i32 range
366                    Literal::BigInteger(big_int_str) => match big_int_str.parse::<i32>() {
367                        Ok(x) => AstExpression::Literal(AstLiteral::Integer(x)),
368                        _ => AstExpression::Literal(self.parse_literal(&a)),
369                    },
370                    _ => AstExpression::Literal(self.parse_literal(&a)),
371                }
372            }
373            Expression::Block(a) => {
374                let ast_block = self.parse_block(&a);
375                AstExpression::Block(self.gc_interface.alloc(ast_block, AllocSiteMarker::Block))
376            }
377        }
378    }

380    pub fn maybe_make_inc_or_dec(&self, local_var_idx: &AstExpression) -> Option<AstExpression> {
381        let (a, b) = match local_var_idx {
382            AstExpression::LocalVarWrite(a, b) => (a, b),
383            _ => unreachable!(),
384        };
385
386        if let AstExpression::BinaryDispatch(message) = &**b {
387            let signature = self.interner.lookup(message.dispatch_node.signature);
388
389            if (signature == "+" || signature == "-") && message.arg == AstExpression::Literal(AstLiteral::Integer(1)) {
390                if let AstExpression::LocalVarRead(local_idx) = message.dispatch_node.receiver {
391                    if local_idx == *a {
392                        match signature {
393                            "+" => {
394                                return Some(AstExpression::IncLocal(*a));
395                            }
396                            "-" => return Some(AstExpression::DecLocal(*a)),
397                            _ => unreachable!(),
398                        }
399                    }
400                }
401            }
402        }
403
404        None
405    }

413    pub fn parse_block(&mut self, blk: &ast::Block) -> AstBlock {
414        let mut locals_set: IndexSet<String> = IndexSet::new(); // TODO: can we do better than allocating whole new IndexSets, for here and the method?..
415        for local in &blk.locals {
416            locals_set.insert(local.clone());
417        }
418
419        let mut args_set: IndexSet<String> = IndexSet::new();
420        args_set.insert("#blockSelf".to_string());
421        for arg in &blk.args {
422            args_set.insert(arg.clone());
423        }
424        self.scopes.push(AstScopeCtxt::init(blk.args.len(), blk.locals.len(), locals_set, args_set));
425
426        let body = self.parse_body(&blk.body);
427        let bl = self.scopes.last().unwrap();
428        let output_blk = AstBlock {
429            nbr_args: bl.get_nbr_args() as u8,
430            nbr_locals: bl.get_nbr_locals() as u8,
431            body,
432        };
433
434        self.scopes.pop();
435        output_blk
436    }

438    pub fn parse_message(&mut self, msg: &ast::Message) -> AstExpression {
439        let msg = {
440            match msg {
441                ast::Message::Regular(reg_msg) => reg_msg,
442                ast::Message::IfInlined(if_inlined_msg) => {
443                    let ast_inlined_node = IfInlinedNode {
444                        expected_bool: if_inlined_msg.expected_bool,
445                        cond_expr: self.parse_expression(&if_inlined_msg.cond_expr),
446                        body_instrs: AstBody {
447                            exprs: if_inlined_msg.body_instrs.iter().map(|e| self.parse_expression(e)).collect(),
448                        },
449                    };
450                    return AstExpression::InlinedCall(Box::new(crate::ast::InlinedNode::IfInlined(ast_inlined_node)));
451                }
452                ast::Message::IfNilInlined(if_nil_inlined_message) => {
453                    let ast_inlined_node = IfNilInlinedNode {
454                        expects_nil: if_nil_inlined_message.expects_nil,
455                        cond_expr: self.parse_expression(&if_nil_inlined_message.cond_expr),
456                        body_instrs: AstBody {
457                            exprs: if_nil_inlined_message.body_instrs.iter().map(|e| self.parse_expression(e)).collect(),
458                        },
459                    };
460                    return AstExpression::InlinedCall(Box::new(crate::ast::InlinedNode::IfNilInlined(ast_inlined_node)));
461                }
462                ast::Message::IfTrueIfFalseInlined(if_true_if_false_inlined_message) => {
463                    let ast_inlined_node = IfTrueIfFalseInlinedNode {
464                        cond_expr: self.parse_expression(&if_true_if_false_inlined_message.cond_expr),
465                        body_1_instrs: AstBody {
466                            exprs: if_true_if_false_inlined_message.body_1_instrs.iter().map(|e| self.parse_expression(e)).collect(),
467                        },
468                        body_2_instrs: AstBody {
469                            exprs: if_true_if_false_inlined_message.body_2_instrs.iter().map(|e| self.parse_expression(e)).collect(),
470                        },
471                        expected_bool: if_true_if_false_inlined_message.expected_bool,
472                    };
473                    return AstExpression::InlinedCall(Box::new(crate::ast::InlinedNode::IfTrueIfFalseInlined(ast_inlined_node)));
474                }
475                ast::Message::IfNilIfNotNilInlined(if_nil_if_not_nil_inlined_message) => {
476                    let ast_inlined_node = IfNilIfNotNilInlinedNode {
477                        cond_expr: self.parse_expression(&if_nil_if_not_nil_inlined_message.cond_expr),
478                        body_1_instrs: AstBody {
479                            exprs: if_nil_if_not_nil_inlined_message.body_1_instrs.iter().map(|e| self.parse_expression(e)).collect(),
480                        },
481                        body_2_instrs: AstBody {
482                            exprs: if_nil_if_not_nil_inlined_message.body_2_instrs.iter().map(|e| self.parse_expression(e)).collect(),
483                        },
484                        expects_nil: if_nil_if_not_nil_inlined_message.expects_nil,
485                    };
486                    return AstExpression::InlinedCall(Box::new(crate::ast::InlinedNode::IfNilIfNotNilInlined(ast_inlined_node)));
487                }
488                ast::Message::WhileInlined(while_inlined_message) => {
489                    let ast_inlined_node = WhileInlinedNode {
490                        expected_bool: while_inlined_message.expected_bool,
491                        cond_instrs: AstBody {
492                            exprs: while_inlined_message.cond_instrs.iter().map(|e| self.parse_expression(e)).collect(),
493                        },
494                        body_instrs: AstBody {
495                            exprs: while_inlined_message.body_instrs.iter().map(|e| self.parse_expression(e)).collect(),
496                        },
497                    };
498
499                    return AstExpression::InlinedCall(Box::new(crate::ast::InlinedNode::WhileInlined(ast_inlined_node)));
500                }
501                ast::Message::AndOrInlined(and_or_inlined_message) => match and_or_inlined_message.is_and {
502                    true => {
503                        let ast_inlined_node = AndInlinedNode {
504                            first: self.parse_expression(&and_or_inlined_message.first),
505                            second: AstBody {
506                                exprs: and_or_inlined_message.second.iter().map(|e| self.parse_expression(e)).collect(),
507                            },
508                        };
509
510                        return AstExpression::InlinedCall(Box::new(crate::ast::InlinedNode::AndInlined(ast_inlined_node)));
511                    }
512                    false => {
513                        let ast_inlined_node = OrInlinedNode {
514                            first: self.parse_expression(&and_or_inlined_message.first),
515                            second: AstBody {
516                                exprs: and_or_inlined_message.second.iter().map(|e| self.parse_expression(e)).collect(),
517                            },
518                        };
519
520                        return AstExpression::InlinedCall(Box::new(crate::ast::InlinedNode::OrInlined(ast_inlined_node)));
521                    }
522                },
523                ast::Message::ToDoInlined(to_do_inlined_message) => {
524                    let accumulator_idx = match self.find_var(&to_do_inlined_message.accumulator_name) {
525                        Some(FoundVar::Local(0, a)) => a as usize,
526                        invalid => panic!("to do inlining couldn't find a valid index for its accumulator: got {:?}", invalid),
527                    };
528
529                    let ast_inlined_node = ToDoInlinedNode {
530                        start: self.parse_expression(&to_do_inlined_message.start_expr),
531                        end: self.parse_expression(&to_do_inlined_message.end_expr),
532                        body: AstBody {
533                            exprs: to_do_inlined_message.body_instrs.iter().map(|e| self.parse_expression(e)).collect(),
534                        },
535                        accumulator_idx,
536                    };
537
538                    return AstExpression::InlinedCall(Box::new(crate::ast::InlinedNode::ToDoInlined(ast_inlined_node)));
539                }
540            }
541        };
542
543        let interned_signature = self.interner.intern(msg.signature.as_str());
544
545        if msg.receiver == Expression::Read(String::from("super")) {
546            return AstExpression::SuperMessage(Box::new(AstSuperMessage {
547                super_class: self
548                    .class
549                    .as_ref()
550                    .unwrap()
551                    .super_class
552                    .clone()
553                    .unwrap_or_else(|| panic!("no super class set, even though the method has a super call?")),
554                signature: interned_signature,
555                values: msg.values.iter().map(|e| self.parse_expression(e)).collect(),
556            }));
557        }
558
559        let receiver = self.parse_expression(&msg.receiver);
560        match msg.values.len() {
561            0 => AstExpression::UnaryDispatch(Box::new(AstUnaryDispatch {
562                dispatch_node: AstDispatchNode {
563                    receiver,
564                    signature: interned_signature,
565                    inline_cache: None,
566                },
567            })),
568            1 => AstExpression::BinaryDispatch(Box::new(AstBinaryDispatch {
569                dispatch_node: AstDispatchNode {
570                    receiver,
571                    signature: interned_signature,
572                    inline_cache: None,
573                },
574                arg: self.parse_expression(msg.values.first().unwrap()),
575            })),
576            2 => AstExpression::TernaryDispatch(Box::new(AstTernaryDispatch {
577                dispatch_node: AstDispatchNode {
578                    receiver,
579                    signature: interned_signature,
580                    inline_cache: None,
581                },
582                arg1: self.parse_expression(msg.values.first().unwrap()),
583                arg2: self.parse_expression(msg.values.get(1).unwrap()),
584            })),
585            _ => AstExpression::NAryDispatch(Box::new(AstNAryDispatch {
586                dispatch_node: AstDispatchNode {
587                    receiver,
588                    signature: interned_signature,
589                    inline_cache: None,
590                },
591                values: msg.values.iter().map(|e| self.parse_expression(e)).collect(),
592            })),
593        }
594    }
595
596    pub(crate) fn global_read(&mut self, name: String) -> AstExpression {
597        if name.as_str() == "super" {
598            return AstExpression::ArgRead((self.scopes.len() - 1) as u8, 0);
599        }
600
601        if self.class.is_none() {
602            return AstExpression::GlobalRead(Box::new(GlobalNode::from(self.interner.intern(name.as_str()))));
603        }
604
605        match self.class.as_ref().unwrap().get_field_offset_by_name(&name) {
606            Some(offset) => AstExpression::FieldRead(offset as u8),
607            _ => AstExpression::GlobalRead(Box::new(GlobalNode::from(self.interner.intern(name.as_str())))),
608        }
609    }
610
611    fn resolve_global_write(&mut self, name: &String, expr: &Expression) -> AstExpression {
612        if self.class.is_none() {
613            panic!(
614                "can't turn the GlobalWrite `{}` into a FieldWrite, and GlobalWrite shouldn't exist at runtime",
615                name
616            );
617        }
618
619        match self.class.as_ref().unwrap().get_field_offset_by_name(name) {
620            Some(offset) => AstExpression::FieldWrite(offset as u8, Box::new(self.parse_expression(expr))),
621            _ => panic!(
622                "can't turn the GlobalWrite `{}` into a FieldWrite, and GlobalWrite shouldn't exist at runtime",
623                name
624            ),
625        }
626    }
627
628    pub(crate) fn parse_literal(&mut self, lit: &ast::Literal) -> AstLiteral {
629        match lit {
630            Literal::String(str) => {
631                let str_ptr = self.gc_interface.alloc(str.clone(), AllocSiteMarker::StringLiteral);
632                AstLiteral::String(str_ptr)
633            }
634            Literal::Symbol(str) => {
635                let interned_sym = self.interner.intern(str.as_str());
636                AstLiteral::Symbol(interned_sym)
637            }
638            Literal::Double(double) => AstLiteral::Double(*double),
639            Literal::Integer(int) => AstLiteral::Integer(*int),
640            Literal::BigInteger(bigint_str) => {
641                let bigint_ptr = self.gc_interface.alloc(bigint_str.parse().unwrap(), AllocSiteMarker::BigInt);
642                AstLiteral::BigInteger(bigint_ptr)
643            }
644            Literal::Array(arr) => {
645                let arr_ptr = {
646                    let arr: Vec<AstLiteral> = arr.iter().map(|lit| self.parse_literal(lit)).collect();
647                    self.gc_interface.alloc_slice(arr.as_slice(), AllocSiteMarker::SliceAstLiteral)
648                };
649                AstLiteral::Array(VecLiteral(arr_ptr))
650            }
651        }
652    }
