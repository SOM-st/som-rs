136#[derive(Debug, Clone, PartialEq)]
137pub struct AstMethodDef {
138    /// The method's signature (eg. `println`, `at:put:` or `==`).
139    pub signature: String,
140    /// The method's body.
141    pub body: AstBody,
142    /// Number of local variables
143    pub locals_nbr: u8,
144}
145
146impl GcType for AstBlock {
147    fn get_magic_gc_id() -> u8 {
148        GcIdentifier::AstBlock as u8
149    }
150
151    fn scan_object(ast_block: Gc<Self>, visit_slot_fn: &mut dyn FnMut(SOMSlot)) {
152        for expr in &ast_block.body.exprs {
153            visit_expr(expr, visit_slot_fn)
154        }
155    }
156
157    fn get_size_in_memory(_self: Gc<Self>) -> usize {
158        size_of::<AstBlock>()
159    }
160}

164impl Display for AstMethodDef {
165    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
166        f.write_fmt(format_args!("Method {} ({} locals):", &self.signature, self.locals_nbr))?;

171impl Display for AstBody {
172    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
173        writeln!(f, "AstBody:")?;
174        for expr in &self.exprs {
175            write!(indented(f), "{}", expr)?;
176        }
177        Ok(())
178    }
179}

181impl Display for AstBlock {

183        writeln!(f, "AstBlock({} args, {} locals):", self.nbr_args, self.nbr_locals)?;
184        for expr in &self.body.exprs {
185            write!(indented(f), "{}", expr)?;
186        }
187        Ok(())
188    }

191// probably not using the indenter lib as one should? though it works. I've not given it much effort
192impl Display for AstExpression {
193    fn fmt(&self, f: &mut Formatter<'_>) -> std::fmt::Result {
194        match self {
195            AstExpression::GlobalRead(global_node) => writeln!(f, "GlobalRead({:?})", global_node.global),
196            AstExpression::LocalVarRead(index) => writeln!(f, "LocalVarRead({})", index),
197            AstExpression::NonLocalVarRead(level, index) => {
198                writeln!(f, "NonLocalVarRead({}, {})", level, index)
199            }
200            AstExpression::ArgRead(level, index) => writeln!(f, "ArgRead({}, {})", level, index),
201            AstExpression::FieldRead(index) => writeln!(f, "FieldRead({})", index),
202            AstExpression::LocalVarWrite(index, expr) => {
203                writeln!(f, "LocalVarWrite({}):", index)?;
204                write!(indented(f), "{}", expr)
205            }
206            AstExpression::NonLocalVarWrite(level, index, expr) => {
207                writeln!(f, "NonLocalVarWrite({}, {}):", level, index)?;
208                write!(indented(f), "{}", expr)
209            }
210            AstExpression::ArgWrite(level, index, expr) => {
211                writeln!(f, "ArgWrite({}, {}):", level, index)?;
212                write!(indented(f), "{}", expr)
213            }
214            AstExpression::FieldWrite(index, expr) => {
215                writeln!(f, "FieldWrite({}):", index)?;
216                write!(indented(f), "{}", expr)
217            }
218            AstExpression::UnaryDispatch(op) => {
219                writeln!(f, "UnaryDispatch \"{:?}\":", op.dispatch_node.signature)?;
220                writeln!(indented(f), "Receiver:")?;
221                write!(indented(&mut indented(f)), "{}", op.dispatch_node.receiver)
222            }
223            AstExpression::BinaryDispatch(op) => {
224                writeln!(f, "BinaryDispatch \"{:?}\":", op.dispatch_node.signature)?;
225                writeln!(indented(f), "Receiver:")?;
226                write!(indented(&mut indented(f)), "{}", op.dispatch_node.receiver)?;
227                writeln!(indented(f), "arg:")?;
228                write!(indented(&mut indented(f)), "{}", op.arg)
229            }
230            AstExpression::TernaryDispatch(op) => {
231                writeln!(f, "TernaryDispatch \"{:?}\":", op.dispatch_node.signature)?;
232                writeln!(indented(f), "Receiver:")?;
233                write!(indented(&mut indented(f)), "{}", op.dispatch_node.receiver)?;
234                writeln!(indented(f), "arg1:")?;
235                write!(indented(&mut indented(f)), "{}", op.arg1)?;
236                writeln!(indented(f), "arg2:")?;
237                write!(indented(&mut indented(f)), "{}", op.arg2)
238            }
239            AstExpression::NAryDispatch(msg) => {
240                writeln!(f, "N-AryDispatch \"{:?}\":", msg.dispatch_node.signature)?;
241                writeln!(indented(f), "Receiver:")?;
242                write!(indented(&mut indented(f)), "{}", msg.dispatch_node.receiver)?;
243                writeln!(indented(f), "Values: {}", if msg.values.is_empty() { "(none)" } else { "" })?;
244                for value in &msg.values {
245                    write!(indented(&mut indented(f)), "{}", value)?;
246                }
247                Ok(())
248            }
249            AstExpression::SuperMessage(msg) => {
250                writeln!(f, "SuperMessage \"{:?}\":", msg.signature)?;
251                writeln!(indented(f), "Receiver: {}", msg.super_class.name)?;
252                writeln!(indented(f), "Values: {}", if msg.values.is_empty() { "(none)" } else { "" })?;
253                for value in &msg.values {
254                    write!(indented(&mut indented(f)), "{}", value)?;
255                }
256                Ok(())
257            }
258            AstExpression::LocalExit(expr) => {
259                writeln!(f, "LocalExit")?;
260                writeln!(indented(f), "{}", expr)
261            }
262            AstExpression::NonLocalExit(expr, index) => {
263                writeln!(f, "NonLocalExit({})", index)?;
264                writeln!(indented(f), "{}", expr)
265            }
266            AstExpression::Literal(literal) => writeln!(f, "Literal({:?})", literal),
267            AstExpression::Block(block) => {
268                writeln!(f, "Block:")?;
269                writeln!(indented(f), "{}", **block)
270            }
271            AstExpression::InlinedCall(inlined_node) => match inlined_node.as_ref() {
272                InlinedNode::IfInlined(node) => writeln!(f, "{}", node),
273                InlinedNode::IfTrueIfFalseInlined(node) => writeln!(f, "{}", node),
274                InlinedNode::IfNilInlined(node) => writeln!(f, "{}", node),
275                InlinedNode::WhileInlined(node) => writeln!(f, "{}", node),
276                InlinedNode::OrInlined(node) => writeln!(f, "{}", node),
277                InlinedNode::AndInlined(node) => writeln!(f, "{}", node),
278                InlinedNode::ToDoInlined(node) => writeln!(f, "{}", node),
279                InlinedNode::IfNilIfNotNilInlined(node) => writeln!(f, "{}", node),
280            },
281            AstExpression::IncLocal(idx) => writeln!(f, "Inc({})", idx),
282            AstExpression::DecLocal(idx) => writeln!(f, "Dec({})", idx),
283        }
284    }
